cmake_minimum_required(VERSION 3.28)
project(LidarFamilyModels LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AlgorithmDependency.cmake")
if(NOT TARGET Lidar::FamilyModels)
    add_library(LidarFamilyModels INTERFACE)
    add_library(Lidar::FamilyModels ALIAS LidarFamilyModels)
    target_compile_features(LidarFamilyModels INTERFACE cxx_std_20)
    target_include_directories(LidarFamilyModels INTERFACE "${CMAKE_CURRENT_LIST_DIR}/../include")
    target_link_libraries(LidarFamilyModels INTERFACE Lidar::Algorithm)
endif()
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    foreach(dependency Qt6::Core Lidar::HandleCore Lidar::DeviceCore Lidar::FamilyCore OpenMP::OpenMP_CXX)
        if(TARGET ${dependency})
            message(FATAL_ERROR "FamilyModels unexpectedly loaded ${dependency}")
        endif()
    endforeach()
    if(BUILD_TESTING)
        add_executable(model_contract ../tests/model_contract.cpp)
        target_link_libraries(model_contract PRIVATE Lidar::FamilyModels)
        add_test(NAME model_contract COMMAND model_contract)
        set(units)
        foreach(header DAQFamily/DAQChannelSpec.h GMBFamily/GimbalModel.h GMBFamily/SiteCalibration.h)
            string(MAKE_C_IDENTIFIER "${header}" unit)
            file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp" "#include <${header}>\n")
            list(APPEND units "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp")
        endforeach()
        add_library(model_headers OBJECT ${units})
        target_link_libraries(model_headers PRIVATE Lidar::FamilyModels)
        foreach(target model_contract model_headers)
            if(MSVC)
                target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
            else()
                target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
            endif()
        endforeach()
    endif()
endif()
