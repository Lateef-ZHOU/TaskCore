add_executable(module_smoke smoke.cpp)
target_link_libraries(module_smoke PRIVATE Lidar::TaskCore)
add_executable(processing_contract processing.cpp)
set_target_properties(processing_contract PROPERTIES AUTOMOC ON)
target_link_libraries(processing_contract PRIVATE Lidar::TaskCore)
if(NOT TASKCORE_WITH_QML AND (TARGET Qt6::Gui OR TARGET Qt6::Qml OR TARGET Qt6::Quick))
    message(FATAL_ERROR "Backend library unexpectedly loaded GUI/QML")
endif()
file(GLOB_RECURSE headers "${PROJECT_SOURCE_DIR}/include/*.h")
if(TARGET Lidar::TaskSupport)
    list(APPEND headers "${PROJECT_SOURCE_DIR}/optional/task-support/include/Task/TaskBase.h")
endif()
if(TARGET Lidar::DAQSaving)
    list(APPEND headers "${PROJECT_SOURCE_DIR}/optional/daq-saving/include/Acquisition/AcqSaver.h")
endif()
if(TARGET Lidar::PcapForms)
    list(APPEND headers "${PROJECT_SOURCE_DIR}/optional/pcap-forms/include/Device/PcapForm.h")
endif()
set(units)
foreach(header IN LISTS headers)
    file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}/include" "${header}")
    string(MAKE_C_IDENTIFIER "${relative}" unit)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp" "#include \"${header}\"\n")
    list(APPEND units "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp")
endforeach()
add_library(public_headers OBJECT ${units})
target_link_libraries(public_headers PRIVATE Lidar::TaskCore)
if(TARGET Lidar::TaskSupport)
    target_link_libraries(public_headers PRIVATE Lidar::TaskSupport)
elseif(TARGET Lidar::DAQSaving)
    target_link_libraries(public_headers PRIVATE Lidar::DAQSaving)
endif()
if(TARGET Lidar::PcapForms)
    target_link_libraries(public_headers PRIVATE Lidar::PcapForms)
endif()
foreach(target module_smoke public_headers processing_contract)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endif()
endforeach()
add_test(NAME module_smoke COMMAND module_smoke)
add_test(NAME processing_contract COMMAND processing_contract)
set_tests_properties(module_smoke processing_contract PROPERTIES TIMEOUT 30
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=null"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")

endif()
