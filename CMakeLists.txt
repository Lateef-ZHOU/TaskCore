cmake_minimum_required(VERSION 3.28)
project(LidarFamilyCore LANGUAGES CXX)
add_subdirectory(models)
find_package(OpenMP REQUIRED)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core)
option(FAMILYCORE_WITH_PCAP_FORMS "Build optional Pcap connection-form adapters" OFF)
if(FAMILYCORE_WITH_PCAP_FORMS)
    set(HANDLECORE_WITH_PCAP ON)
endif()
include(cmake/Dependencies.cmake)
if(NOT TARGET Lidar::DeviceCore OR NOT TARGET Lidar::Algorithm)
    message(FATAL_ERROR "FamilyCore requires Lidar::DeviceCore and Lidar::Algorithm")
endif()
add_library(LidarFamilyCore STATIC
    include/DAQFamily/AcqSaver.h
    include/DAQFamily/DAQDevice.h
    include/GMBFamily/GMBDevice.h)
set_target_properties(LidarFamilyCore PROPERTIES AUTOMOC ON)
qt_extract_metatypes(LidarFamilyCore)
if(MSVC)
    target_compile_options(LidarFamilyCore PRIVATE /W4)
endif()
add_library(Lidar::FamilyCore ALIAS LidarFamilyCore)
target_include_directories(LidarFamilyCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(LidarFamilyCore PUBLIC Lidar::DeviceCore Lidar::FamilyModels OpenMP::OpenMP_CXX Boost::circular_buffer)

option(FAMILYCORE_WITH_TASK_SUPPORT "Build the optional acquisition-task business base" OFF)
option(FAMILYCORE_WITH_QML "Build optional family controls and task reference pages" OFF)
if(FAMILYCORE_WITH_TASK_SUPPORT OR FAMILYCORE_WITH_QML)
    add_subdirectory(optional/task-support)
endif()
if(FAMILYCORE_WITH_QML)
    add_subdirectory(optional/qml)
endif()

if(FAMILYCORE_WITH_PCAP_FORMS)
    add_subdirectory(optional/pcap-forms)
endif()

# 作为依赖时不接管宿主的测试开关。
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory(tests)
    endif()
endif()
